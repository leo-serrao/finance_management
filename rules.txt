rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Helpers
    function isSignedIn() {
      return request.auth != null;
    }

    // SHARED GROUPS
    match /sharedGroups/{groupId} {

      allow create: if isSignedIn()
        && request.resource.data.keys().hasAll(['name', 'members'])
        && request.resource.data.name is string
        && request.resource.data.members is list
        && request.resource.data.members.size() > 0
        && request.auth.uid in request.resource.data.members;

      allow read: if isSignedIn()
        && request.auth.uid in resource.data.members;

      allow update, delete: if isSignedIn()
        && request.auth.uid in resource.data.members;

      match /expenses/{expenseId} {

        allow read, create, update, delete: if isSignedIn()
          && request.auth.uid in get(
            /databases/$(database)/documents/sharedGroups/$(groupId)
          ).data.members;
      }
      
      match /debtSettlements/{settlementId} {
        // Members of the group can read and create settlement payments.
        allow read: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/sharedGroups/$(groupId)).data.members;
        allow create: if isSignedIn()
          && request.auth.uid in get(/databases/$(database)/documents/sharedGroups/$(groupId)).data.members
          && request.resource.data.keys().hasAll(['fromUserId','toUserId','amount','groupId'])
          && request.resource.data.fromUserId is string
          && request.resource.data.toUserId is string
          && request.resource.data.amount is number
          && request.resource.data.amount > 0
          && request.resource.data.groupId == groupId;
        // Allow deletion of a payment by either the payer (`fromUserId`) or the payee (`toUserId`), members only.
        allow delete: if isSignedIn()
          && request.auth.uid in get(/databases/$(database)/documents/sharedGroups/$(groupId)).data.members
          && (request.auth.uid == resource.data.fromUserId || request.auth.uid == resource.data.toUserId);
        // Do not allow client-side updates to settlement records.
        allow update: if false;
      }
    }

    // USERS (adjust privacy as needed)
    match /users/{userId} {
      allow read: if isSignedIn();
      allow write: if request.auth != null && request.auth.uid == userId;

      // fixed/variable/savingBoxes rules (unchanged)
      match /fixedExpenses/{expenseId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create, update: if request.auth != null && request.auth.uid == userId
          && request.resource.data.keys().hasAll(['name','amount'])
          && request.resource.data.name is string
          && request.resource.data.amount is number
          && request.resource.data.amount > 0;
        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      match /variableExpenses/{expenseId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create, update: if request.auth != null && request.auth.uid == userId
          && request.resource.data.keys().hasAll(['title','amount','date'])
          && request.resource.data.title is string
          && request.resource.data.amount is number
          && request.resource.data.amount >= 0
          && request.resource.data.date is string;
        allow delete: if request.auth != null && request.auth.uid == userId;
      }

      match /savingBoxes/{boxId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create, update: if request.auth != null && request.auth.uid == userId
          && request.resource.data.keys().hasAll(['name','amount'])
          && request.resource.data.name is string
          && request.resource.data.amount is number
          && request.resource.data.amount >= 0
          && ( !('emoji' in request.resource.data) || request.resource.data.emoji is string )
          && ( !('color' in request.resource.data) || request.resource.data.color is string );
        allow delete: if request.auth != null && request.auth.uid == userId;
      }
    }
  }
}
